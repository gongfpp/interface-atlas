# 界面图谱抖音宣传片｜Remotion 实现指南

## 快速启动

### 前置环境
- Node.js 18+
- Remotion CLI (`npm install -g remotion`)
- video-shotcraft 模板框架已准备在 `/tmp/atlas-promo/`

### 工作流
```bash
# 1. 复制模板到项目
cp -r /tmp/atlas-promo ./video-src

# 2. 安装依赖
cd video-src && npm install

# 3. 实现核心镜头代码（见下方代码框架）

# 4. 开发预览
npm run dev  # 启动 Remotion 开发服务器

# 5. 静帧验收
npx remotion still src/index.ts AtlasPromo out/qa/shot-1.png --frame=45

# 6. 全片渲染
npx remotion render src/index.ts AtlasPromo out/atlas-promo-30s.mp4

# 7. 生成无 BGM 版本（见音频章节）
```

## 核心实现代码框架

### 1. src/Root.tsx - 主入口

```tsx
import React from 'react';
import { Composition } from 'remotion';
import { AtlasPromo } from './AtlasPromo';

export const MyComposition = () => (
  <Composition
    id="AtlasPromo"
    component={AtlasPromo}
    durationInFrames={900}  // 30s @ 30fps
    fps={30}
    width={1080}  // 竖版
    height={1920}
    defaultProps={{
      bgm: true,  // 控制 BGM 开关
    }}
  />
);
```

### 2. src/AtlasPromo.tsx - 主镜头组件

```tsx
import React from 'react';
import {
  AbsoluteFill,
  Sequence,
  useVideoConfig,
  Img,
  Audio,
  OffthreadVideo,
} from 'remotion';
import { SHOTS } from './timeline';

interface Props {
  bgm: boolean;
}

export const AtlasPromo: React.FC<Props> = ({ bgm = true }) => {
  const { fps, durationInFrames } = useVideoConfig();

  return (
    <AbsoluteFill style={{ backgroundColor: '#1F2937' }}>
      {/* 背景 */}
      <AbsoluteFill style={{ background: 'linear-gradient(135deg, #1F2937 0%, #111827 100%)' }} />

      {/* 各镜头序列 */}
      {SHOTS.map((shot, idx) => (
        <Sequence key={idx} from={shot.from} durationInFrames={shot.duration}>
          <shot.Component {...shot.props} />
        </Sequence>
      ))}

      {/* BGM 音频轨 */}
      {bgm && (
        <Audio
          src={require('./assets/audio/bgm-tech-house.mp3')}
          volume={0.34}
        />
      )}
    </AbsoluteFill>
  );
};
```

### 4. src/shots/Shot1_PainPoint.tsx - 镜头 1 实现

```tsx
import React from 'react';
import { AbsoluteFill, interpolate, spring, useCurrentFrame, useVideoConfig } from 'remotion';
import { BRAND_GREEN } from '../theme';

// ScrambleDecode 参数
const CHARSET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';

const mulberry32 = (seed: number) => {
  return () => {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) >>> 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) >>> 0;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
};

export const Shot1_PainPoint: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  // 文本：「和 AI 说不清需求」(13 字)
  const text = '和 A I 说 不 清 需 求';
  const chars = text.split('');

  // 两段时间：乱码 (0-20f) + 解码 (20-90f)
  const decodeStart = 20;
  const decodeEnd = 90;

  return (
    <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center' }}>
      <div style={{
        display: 'flex',
        gap: '4px',
        fontFamily: 'SF Mono, Monaco, monospace',
        fontSize: 64,
        fontWeight: 600,
        color: BRAND_GREEN,
        textAlign: 'center',
      }}>
        {chars.map((char, i) => {
          const rng = mulberry32(i * 101 + 13);
          let displayChar = char;
          let opacity = 1;
          let rotateZ = 0;

          if (frame < decodeStart) {
            // 乱码阶段
            displayChar = CHARSET[Math.floor(rng() * CHARSET.length)];
          } else if (frame < decodeStart + i * 6) {
            // 该字符还未解码
            displayChar = CHARSET[Math.floor(rng() * CHARSET.length)];
          } else {
            // 已解码
            displayChar = char;
            const decodeProgress = Math.min(1, (frame - decodeStart - i * 6) / 2);
            opacity = decodeProgress > 0.8 ? 1 : 0;  // 闪现效果
          }

          return (
            <span
              key={i}
              style={{
                opacity,
                display: 'inline-block',
                width: '1ch',
                transition: 'opacity 0.1s',
              }}
            >
              {displayChar}
            </span>
          );
        })}
      </div>

      {/* × × 脉冲 */}
      <div style={{
        position: 'absolute',
        top: '60%',
        display: 'flex',
        gap: '20px',
        fontSize: 48,
        color: '#EF4444',
        opacity: spring({ frame: frame - 50, fps, config: { damping: 10 } }),
      }}>
        <span>×</span>
        <span>×</span>
      </div>
    </AbsoluteFill>
  );
};
```

### 5. src/shots/Shot2_Search.tsx - 镜头 2 实现

```tsx
import React, { useMemo } from 'react';
import { AbsoluteFill, interpolate, useCurrentFrame } from 'remotion';
import { BRAND_GREEN } from '../theme';

interface Props {
  searchText: string;
}

export const Shot2_Search: React.FC<Props> = ({ searchText }) => {
  const frame = useCurrentFrame();

  // 搜索框入场 (0-30f)
  const boxScale = interpolate(frame, [0, 30], [0.8, 1], { extrapolateLeft: 'clamp' });
  const boxOpacity = interpolate(frame, [0, 20], [0, 1], { extrapolateLeft: 'clamp' });

  // 文字输入 (40-110f)
  const inputProgress = Math.max(0, Math.min(1, (frame - 40) / 70));
  const visibleChars = Math.floor(inputProgress * searchText.length);
  const inputedText = searchText.slice(0, visibleChars);

  return (
    <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center' }}>
      <div style={{
        width: 600,
        padding: '20px 24px',
        backgroundColor: 'rgba(255, 255, 255, 0.05)',
        border: `2px solid ${BRAND_GREEN}`,
        borderRadius: 12,
        fontSize: 32,
        color: '#FFFFFF',
        fontFamily: '-apple-system, BlinkMacSystemFont, "Segoe UI"',
        transform: `scale(${boxScale})`,
        opacity: boxOpacity,
        boxShadow: `0 0 24px ${BRAND_GREEN}4D`,
      }}>
        🔍 {inputedText}
        {frame > 40 && frame < 110 && <span style={{ animation: 'blink 0.5s infinite' }}>|</span>}
      </div>

      {/* 结果提示 */}
      <div style={{
        position: 'absolute',
        top: '60%',
        fontSize: 20,
        color: BRAND_GREEN,
        opacity: interpolate(frame, [80, 120], [0, 1]),
      }}>
        ✓ Skeleton Loading
      </div>
    </AbsoluteFill>
  );
};
```

## 音频策略

### BGM 选择

推荐：tech-house, ~110 BPM, 强鼓点
- 候选：assets/audio/bgm/ 中的 `techno-pulse-110bpm.mp3`
- 参数：volume 0.34（给 SFX 留 headroom）

### SFX 钉帧

按 `SFX_CUES` 表在 src/SFXLayer.tsx 中实现：

```tsx
export const SFXLayer: React.FC = () => {
  return (
    <AbsoluteFill>
      {SFX_CUES.map((cue) => (
        <Sequence key={cue.frame} from={cue.frame} durationInFrames={30}>
          <Audio
            src={require(`./assets/audio/sfx/${cue.file}.mp3`)}
            volume={0.8}
          />
        </Sequence>
      ))}
    </AbsoluteFill>
  );
};
```

### 双版本输出

在 `remotion.config.ts` 中定义：

```ts
export const inputProps = {
  bgm: [true, false],  // 生成两个 Composition：带 BGM + 无 BGM
};
```

渲染时：
```bash
npx remotion render src/index.ts AtlasPromo out/atlas-with-bgm.mp4 --props='{"bgm":true}'
npx remotion render src/index.ts AtlasPromo out/atlas-sfx-only.mp4 --props='{"bgm":false}'
```
